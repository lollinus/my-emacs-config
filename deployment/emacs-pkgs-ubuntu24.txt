# System packages required to build Emacs 31.x from source on Ubuntu 24.04 LTS
# Install with: sudo apt install $(grep -Ev '^\s*(#|$)' emacs-pkgs-ubuntu24.txt)
#
# Verify after configure that nothing was silently dropped:
#   grep -E 'HAVE_(RSVG|GIF|SQLITE3|LIBXML2|LCMS2|XPM|TREE_SITTER|NATIVE_COMP) ' src/config.h
# Emacs' configure turns a missing optional library into a disabled feature
# instead of an error, even when --with-<feature> is passed explicitly.

# Core build tools
build-essential
autoconf
automake
pkgconf
texinfo
clang
git
curl
xz-utils

# Native compilation (libgccjit required for --with-native-compilation=aot)
# The build locates libgccjit.h under /usr/lib/gcc/x86_64-linux-gnu/<ver>/
# (picking the highest version), so the libgccjit-NN-dev version need not
# match the default gcc.  libgccjit-14-dev pulls in libgcc-14-dev, which
# provides the crt/libgcc files libgccjit needs at runtime.
libgccjit0
libgccjit-14-dev

# Tree-sitter (required for --with-tree-sitter)
# Note: libtree-sitter-dev is intentionally omitted — both build paths
# (emacs-config-ubuntu24.sh and the Bazel build) compile libtree-sitter
# from source to get ABI-15 support. Ubuntu 24.04's package (0.20.8) only
# supports ABI ≤ 14, which is incompatible with grammar files generated by
# tree-sitter CLI 0.23+.

# GUI / display
# libgtk-3-dev  — GTK toolkits (--with-pgtk, --with-x-toolkit=gtk3)
# libxaw7-dev   — Lucid (Athena) toolkit (--with-x-toolkit=lucid); Bazel X11 default
# libxt-dev     — X Toolkit Intrinsics; required by Lucid toolkit
libgtk-3-dev
libxaw7-dev
libxt-dev
libdbus-1-dev

# Text shaping / font rendering
libharfbuzz-dev
libharfbuzz-bin

# Image format support
libjpeg-dev
libpng-dev
libtiff-dev
libgif-dev
libxpm-dev
librsvg2-dev
libwebp-dev
liblcms2-dev

# TLS / security
# (gnutls-dev is a virtual name with no installation candidate on 24.04)
libgnutls28-dev

# SQLite (for org-roam, consult, etc.)
libsqlite3-dev

# XML (eww/shr HTML rendering, libxml-parse-*) / misc
libxml2-dev
libncurses-dev
libacl1-dev

# Sound (optional but commonly needed)
libasound2-dev

# vterm terminal emulator support
libvterm-dev

# Note: libjansson-dev is no longer needed — Emacs 30+ has a built-in JSON parser.
